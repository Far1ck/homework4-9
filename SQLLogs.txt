Получение всех клиентов с использованием JOIN FETCH:
Hibernate:
    select
        c1_0.id,
        a1_0.client_id,
        a1_0.id,
        a1_0.account_number,
        a1_0.balance,
        c1_0.email,
        c1_0.name
    from
        clients c1_0
    join
        accounts a1_0
            on c1_0.id=a1_0.client_id
